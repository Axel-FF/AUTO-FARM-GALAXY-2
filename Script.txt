	hue = (hue + 0.55) % 360
	local c = Color3.fromHSV(hue / 360, 0.9, 1)
	for _, l in pairs(rgbLabels) do
		l.TextColor3 = c
	end
	Stroke.Color = c
	SpeedBar.BackgroundColor3 = c

local farming = false
local noclip = false
local minimized = false
local FarmSpeed = 50

local function UpdateSpeedUI()
	SpeedBox.Text = tostring(FarmSpeed)
	SpeedBar.Size = UDim2.new(FarmSpeed / 100, 0, 1, 0)
end

local function GetRoot()
	local char = LP.Character
	if not char then return nil end
	return char:FindFirstChild("HumanoidRootPart")
end

local function ApplyNoclip()
	local char = LP.Character
	if not char then return end
	for _, p in pairs(char:GetDescendants()) do
		if p:IsA("BasePart") then
			p.CanCollide = not noclip
		end
	end
end

RunService.Stepped:Connect(function()
	if noclip then ApplyNoclip() end
end)

local function GoTo(pos)
	local root = GetRoot()
	if not root then return end
	local duration = math.clamp(1.2 - (FarmSpeed / 100) * 1.12, 0.08, 1.2)
	local goal = CFrame.new(pos + Vector3.new(0, 3, 0))
	local tw = TweenService:Create(root, TweenInfo.new(duration, Enum.EasingStyle.Linear), {CFrame = goal})
	tw:Play()
	tw.Completed:Wait()
end

local function FarmLoop()
	while farming do
		for i, pos in ipairs(Path) do
			if not farming then break end
			PointLbl.Text = "Punto: " .. i .. " / " .. #Path
			Status.Text = "Estado: Farmeando..."
			GoTo(pos)
		end
		if farming then
			Status.Text = "Estado: Reiniciando..."
			task.wait(0.25)
		end
	end
	Status.Text = "Estado: OFF"
	PointLbl.Text = "Punto: 0 / " .. #Path
end

FarmBtn.MouseButton1Click:Connect(function()
	farming = not farming
	FarmBtn.Text = farming and "FARM   ON" or "FARM   OFF"
	FarmBtn.BackgroundColor3 = farming and Color3.fromRGB(25, 90, 50) or Color3.fromRGB(35, 35, 50)
	if farming then
		Status.Text = "Estado: ON"
		if not noclip then
			noclip = true
			NoclipBtn.Text = "NOCLIP   ON"
			NoclipBtn.BackgroundColor3 = Color3.fromRGB(25, 90, 50)
		end
		task.spawn(FarmLoop)
	else
		Status.Text = "Estado: OFF"
	end
end)

NoclipBtn.MouseButton1Click:Connect(function()
	noclip = not noclip
	NoclipBtn.Text = noclip and "NOCLIP   ON" or "NOCLIP   OFF"
	NoclipBtn.BackgroundColor3 = noclip and Color3.fromRGB(25, 90, 50) or Color3.fromRGB(35, 35, 50)
	ApplyNoclip()
end)

SpeedDn.MouseButton1Click:Connect(function()
	FarmSpeed = math.max(0, FarmSpeed - 5)
	UpdateSpeedUI()
end)

SpeedUp.MouseButton1Click:Connect(function()
	FarmSpeed = math.min(100, FarmSpeed + 5)
	UpdateSpeedUI()
end)

	Body.Visible = not minimized
	MinBtn.Text = minimized and "+" or "-"
end)

Close.MouseButton1Click:Connect(function()
	farming = false

end)
